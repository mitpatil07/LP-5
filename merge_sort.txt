#include <iostream>
#include <vector>
#include <omp.h>

using namespace std;

// Merge function to combine two sorted halves
void merge(vector<int> &arr, int st, int mid, int end) {
    int n1 = mid - st + 1;
    int n2 = end - mid;
    vector<int> L(n1), R(n2);

    for (int i = 0; i < n1; i++) L[i] = arr[st + i];
    for (int j = 0; j < n2; j++) R[j] = arr[mid + 1 + j];

    int i = 0, j = 0, k = st;
    while (i < n1 && j < n2) {
        if (L[i] <= R[j]) arr[k++] = L[i++];
        else arr[k++] = R[j++];
    }
    while (i < n1) arr[k++] = L[i++];
    while (j < n2) arr[k++] = R[j++];
}

// Sequential Merge Sort
void seqMergeSort(vector<int> &arr, int st, int end) {
    if (st < end) {
        int mid = st + (end - st) / 2;
        seqMergeSort(arr, st, mid);
        seqMergeSort(arr, mid + 1, end);
        merge(arr, st, mid, end);
    }
}

// Parallel Merge Sort using OpenMP Sections
void parMergeSort(vector<int> &arr, int st, int end) {
    if (st < end) {
        int mid = st + (end - st) / 2;

        #pragma omp parallel sections
        {
            #pragma omp section
            parMergeSort(arr, st, mid);

            #pragma omp section
            parMergeSort(arr, mid + 1, end);
        }
        merge(arr, st, mid, end);
    }
}

int main() {
    int n;
    cout << "--- Parallel Merge Sort ---\n";
    cout << "Enter number of elements: ";
    cin >> n;

    vector<int> a(n);
    vector<int> b(n);

    cout << "Enter " << n << " elements:\n";
    for (int i = 0; i < n; i++) {
        cin >> a[i];
        b[i] = a[i]; // Copy for parallel sort
    }

    // Sequential Sorting
    double start = omp_get_wtime();
    seqMergeSort(a, 0, n - 1);
    double end = omp_get_wtime();
    cout << "\nSequential Merge Sort Time: " << end - start << " seconds" << endl;

    // Parallel Sorting
    start = omp_get_wtime();
    parMergeSort(b, 0, n - 1);
    end = omp_get_wtime();
    cout << "Parallel Merge Sort Time: " << end - start << " seconds" << endl;

    cout << "\nSorted Array: ";
    for (int i = 0; i < n; i++) {
        cout << b[i] << " ";
    }
    cout << endl;

    return 0;
}
