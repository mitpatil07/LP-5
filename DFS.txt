#include <iostream>
#include <vector>
#include <omp.h>

using namespace std;

vector<vector<int>> graph;
vector<int> visited;

void dfs(int node) {
    visited[node] = 1;
    cout << node << " ";

    for (int i = 0; i < graph[node].size(); i++) {
        int next = graph[node][i];

        if (!visited[next]) {
            #pragma omp task
            {
                if (!visited[next]) {
                    dfs(next);
                }
            }
        }
    }
}

int main() {
    int n, e;

    cout << "Enter total number of vertices: ";
    cin >> n;
    cout << "Enter total number of edges: ";
    cin >> e;

    graph.resize(n);
    visited.resize(n, 0);

    cout << "Enter edges (source destination):" << endl;
    for (int i = 0; i < e; i++) {
        int u, v;
        cin >> u >> v;
        if (u >= 0 && u < n && v >= 0 && v < n) {
            graph[u].push_back(v);
            graph[v].push_back(u);
        }
    }

    cout << "\nParallel DFS Traversal starting from node 0: ";

    #pragma omp parallel
    {
        #pragma omp single
        {
            dfs(0);
        }
    }

    cout << endl;
    return 0;
}
